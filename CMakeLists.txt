cmake_minimum_required(VERSION 3.16)
project(16WPMSP LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()
add_library(16wpmsp INTERFACE)
target_include_directories(16wpmsp INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_options(16wpmsp INTERFACE -march=znver4 -O3)
find_package(Threads REQUIRED)
target_link_libraries(16wpmsp INTERFACE Threads::Threads)
add_executable(bench_16wpmsp bench/bench_16wpmsp.cpp)
target_link_libraries(bench_16wpmsp PRIVATE 16wpmsp)
set(EXT ${CMAKE_CURRENT_SOURCE_DIR}/external)
foreach(dep pdqsort ips4o highway ska_sort x86-simd-sort)
    if(EXISTS ${EXT}/${dep})
        target_include_directories(bench_16wpmsp PRIVATE ${EXT}/${dep})
    endif()
endforeach()
find_package(TBB QUIET)
if(TBB_FOUND)
    target_link_libraries(bench_16wpmsp PRIVATE TBB::tbb)
endif()
add_executable(stress_16wpmsp tests/stress_16wpmsp.cpp)
target_link_libraries(stress_16wpmsp PRIVATE 16wpmsp)
enable_testing()
add_test(NAME stress COMMAND stress_16wpmsp)
